cmake_minimum_required(VERSION 3.15)
project(unity_astc_quality LANGUAGES CXX)
if(NOT DEFINED METRICS_SOURCE)
    message(FATAL_ERROR "METRICS_SOURCE must point to the locally extracted Unity functions")
endif()
add_library(unity_astc_quality SHARED "${METRICS_SOURCE}")
target_compile_features(unity_astc_quality PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(unity_astc_quality PRIVATE /fp:precise /EHsc)
endif()
